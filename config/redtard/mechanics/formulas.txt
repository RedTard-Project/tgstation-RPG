# Formulas: name = expression. Functions: min, max, clamp, abs, round, floor, ceil, sqrt, if. Operators: + - * / % ^ < <= > >= == != and or not.

# Damage multiplier from the attacker's strength (Ratwood). min_strength is the weapon's need, halved when wielded, 0 if none. Inputs: strength, min_strength.
strength_factor = if(strength > 14, 1.4 + (strength - 14) * 0.034, 1 + (strength - 10) * 0.1) * if(min_strength > strength, 0.3, 1)

# Parry value of one weapon or of bare hands (Azure Peak): weapon skill rank and weapon defense. Inputs: skill, defense.
weapon_defense = skill * 20 + defense * 10

# Parry penalty when a faster attacker uses a swift weapon or bare hands (Azure Peak); perception and intelligence offset it. Inputs: swift_attack, attacker_armed, attacker_speed, defender_speed, attacker_perception, defender_perception, attacker_intelligence, defender_intelligence, cap.
parry_speed_pressure = if(swift_attack and attacker_speed > defender_speed, clamp((attacker_speed - defender_speed) * 10 - max(0, (defender_perception - attacker_perception) * if(attacker_armed, 5, 10)) - max(0, (defender_intelligence - attacker_intelligence) * 3), 0, cap), 0)

# Parry chance in percent (Azure Peak). defense is the best weapon_defense; swift_guard is a lone swift weapon in light armor. Inputs: defense, attacker_skill, running, swift_guard, speed_pressure, prone, dual_wield, untrained_armor, bound.
parry_chance = clamp((defense - attacker_skill * 20 - running * 15 + swift_guard * 10 - speed_pressure) * if(prone, 0.65, 1) - dual_wield * 5 + bound * 20, 5, if(untrained_armor, 75, 90))

# Dodge chance in percent (Azure Peak). base is the attacker's defprob (dodge_base for a person); defender_skill is the defender's rank in the attacker's skill; improvised is a weapon without a skill. Inputs: defender_speed, attacker_speed, swift_attack, heavy_attack, improvised, defender_skill, attacker_skill, prone, dodge_expert, dual_wield, fatigue, base.
dodge_chance = clamp(if(prone, 0.25, 1) * if(dodge_expert and not prone, 90, base + (defender_speed - attacker_speed) * 10 - if(swift_attack and attacker_speed > defender_speed, (attacker_speed - defender_speed) * 10, 0) + if(heavy_attack and defender_speed > attacker_speed, (defender_speed - attacker_speed) * 10, 0) + if(improvised, 10, (defender_skill - attacker_skill) * 10)) + fatigue - dual_wield * 5, 5, 90 + fatigue)

# Percent of damage the best covering armor stops (Ratwood). Inputs: rating, penetration.
armor_protection = clamp(rating - penetration, 0, 100)

# Integrity the armor loses from one hit (Ratwood). incoming is the damage before armor. Inputs: incoming, rating, penetration, blunt_attack, blunt_rating.
armor_wear = if(blunt_attack, incoming - incoming * max(blunt_rating, 0) / 200, if(incoming + penetration > rating, if(penetration >= rating, incoming / 2, max(rating - penetration, incoming * 0.4)), incoming))

# Wound chance multiplier from constitution. Inputs: constitution.
wound_constitution = clamp(1 - (constitution - 10) * 0.03, 0.5, 1.5)

# Feint success chance in percent (Azure Peak): weapon skill ranks and intelligence of both sides. A target that is already open cannot be feinted again. Inputs: skill, target_skill, intelligence, target_intelligence, target_vulnerable.
feint_chance = if(target_vulnerable, 0, clamp(50 + (skill - target_skill) * 15 + (intelligence - target_intelligence) * 10, 10, 90))

# How long looking around takes, in deciseconds (Azure Peak). Inputs: perception, tracking.
look_around_time = clamp(50 - perception * 2 - tracking * 5, 7, 50)

# Chance in percent that looking around finds a hidden creature (Azure Peak). target_stealth is the best of its sneaking, holy and arcane ranks, 0 without a mind. Inputs: perception, tracking, target_stealth, target_perception.
look_around_chance = max(perception * 2 + tracking * 5 - target_stealth * 5 - if(target_perception > 10, target_perception / 2, 0), 5)

# Maximum energy (blue stamina) from willpower and the athletics rank (Azure Peak). Inputs: willpower, athletics.
energy_max = (willpower + athletics / 2) * 100

# Pick swings needed to break a rock wall (Ratwood): wall integrity, the pick's force and the mining rank. One swing takes a melee click delay. Inputs: integrity, force, mining.
mining_swings = ceil(integrity / max(force * (8 + mining * 1.5), 1))

# Time of one attempt to pry a window with a knife, in deciseconds (Ratwood). Inputs: lockpicking, perception, speed, picklvl, open.
window_pry_time = clamp(100 - (lockpicking * 8 + perception / 4 * 5 + speed / 4 * 5 + picklvl * 2), 20, 70)

# Progress of one successful attempt; the window gives at 100 (reinforced 150) (Ratwood). Inputs: lockpicking, perception, speed, picklvl, open.
window_pry_progress = clamp(10 + lockpicking * 6 + perception / 4 + speed / 8 + picklvl * 2, 10, 100)

# Chance in percent that one attempt succeeds; a failure breaks the window (Ratwood). picklvl is the knife's lockpicking level. Inputs: lockpicking, perception, speed, picklvl, open.
window_pry_chance = clamp((10 + lockpicking * 11 + perception / 4 * 4 + speed / 4 * 3) * picklvl * if(open, 1.1, 1), 1, 96)

# Time to arm a floor trap by hand, in deciseconds (Ratwood). Needs crafting 1 or more. Inputs: crafting.
trap_set_time = max((8 - max(crafting * 2, 2)) * 10, 0)

# Time to disarm a floor trap by hand, in deciseconds (Ratwood). Needs crafting 1 or more. Inputs: crafting.
trap_disarm_time = max((14 - max(crafting * 2, 2)) * 10, 0)

# Time to spring open or shut a mantrap held in hand, in deciseconds (Ratwood). Inputs: strength, crafting.
mantrap_set_time = max(50 - strength * 2, 0)

# Chance in percent to arm or shut a mantrap held in hand (Ratwood). Inputs: strength, crafting.
mantrap_set_chance = clamp(10 + crafting * 15, 0, 100)

# Time to disarm an armed mantrap on the ground, in deciseconds (Ratwood); an interruption springs it on the hand. Inputs: crafting.
mantrap_disarm_time = max(100 - max(crafting * 20, 20), 0)

# Whether looking around marks a hiding place with someone tucked inside (Azure Peak): non-zero means found. Inputs: perception, target_sneaking.
look_around_spot = perception >= 8 + target_sneaking * 2

# Bare-handed damage before the strike formula: a punch, twisting, chewing, headbutting (Ratwood get_punch_dmg). civilized is the civilized barbarian trait. Inputs: strength, civilized.
unarmed_damage = (17 + civilized * 5) * if(strength > 14, 1.4 + (strength - 14) * 0.034, 1 + (strength - 10) * 0.1)

# Chance in percent to tighten a grab, or to shove a grabbed foe down (Ratwood). Strength and wrestling are grabber minus target; noise is -5 to 5; modifier is 2 for a compliant target, +0.25 restrained, +0.05 lying, +-0.3 for combat readiness, +0.15 chokehold. Inputs: strength_diff, skill_diff, noise, modifier.
grab_chance = clamp(((4 + strength_diff / 2 + skill_diff) * 10 + noise) * modifier, 5, 95)

# Chance in percent to take the item from a grabbed hand (Ratwood); 20 more to knock it to the floor instead. weapon_skill is the holder's rank in the item's skill. Inputs: strength_diff, skill_diff, modifier, weapon_skill, wielded.
grab_disarm_chance = clamp((3 + strength_diff / 4 + skill_diff) * 10 * modifier, 5, 95) - weapon_skill * 5 - wielded * 20

# How long one pin holds a grabbed foe on the ground, in deciseconds (Ratwood). Inputs: strength_diff, skill_diff, modifier.
grab_pin_time = max((65 + skill_diff * 10 + strength_diff * 5) * modifier, 20)

# Damage of smashing a grabbed limb into a wall, floor or table (Ratwood). punch is unarmed_damage, unarmed the unarmed skill rank. Inputs: punch, unarmed.
grab_smash_damage = punch * (1 + unarmed / 10)

# Suffocation per choke of a neck grab (Ratwood). chokehold: grabbed from behind; held_back: the victim holds the choker aggressively. Inputs: strength, chokehold, held_back.
choke_damage = if(strength > 14, 14, strength * 0.75) * if(chokehold, 1.2, 1) * if(held_back, 0.95, 1)

# Damage of the first bite (Ratwood); it only hurts a limb that is already wounded unless the biter has a strong bite. Inputs: strength, strong_bite.
bite_damage = strength / 2 * if(strong_bite, 2, 1)

# Chance in percent that a blow lands on the aimed body part (Azure Peak resolve_aimed_zone). major_zone: a whole limb, head or chest; face_zone: head or face against someone with a mind; grab_state: 1 held, 2 held aggressively; low_aim: the attacker lies and aims at legs or feet; blunt_precise: a blunt blow at a small part; short_weapon: short weapon or bare hands. A miss falls on the whole limb or the chest. Inputs: major_zone, face_zone, perception, grab_state, target_prone, target_open, low_aim, weapon_skill, stab, pick, cut, blunt_precise, short_weapon, aimed, swift.
strike_accuracy = clamp(major_zone * 10 - face_zone * 24 + if(perception > 10, min((perception - 10) * 8, 40), 0) - if(perception < 10, (10 - perception) * 10, 0) + if(grab_state >= 2, 20, if(grab_state >= 1, 10, 0)) + target_prone * 30 + target_open * 20 + low_aim * 5 + weapon_skill * 8 + stab * 10 + pick * 15 + cut * 6 - blunt_precise * 10 + short_weapon * 10 + aimed * 20 - swift * 20, 5, 95)

# Damage of one blow before armor (Azure Peak get_complex_damage). force is the weapon's force or bare-handed damage, strength_factor the strength_factor formula (1 for bare hands), damage_factor the intent's, stance_factor the alt intent's; prone: the attacker lies; edge: share of the strength and intent bonuses a worn blade keeps (1 for a sharp or blunt weapon); chipped: factor of a badly chipped blade (Azure Peak sharpness tiers). Inputs: force, strength_factor, damage_factor, stance_factor, prone, edge, chipped, uncharged.
strike_damage = max(round(force * if(strength_factor > 1, 1 + (strength_factor - 1) * edge, strength_factor) * if(damage_factor > 1, 1 + (damage_factor - 1) * edge, damage_factor) * stance_factor * if(prone, 0.5, 1) * chipped * if(uncharged, 0.5, 1)), 1)

# Chance in percent of one critical wound (Azure Peak try_crit). limb_damage: share of the limb's damage; divisor: crit_damage_divisor or crit_chest_divisor; bonus: 1 for a matching alt intent (strong for fractures and chops, aimed for cuts and stabs). Inputs: limb_damage, damage, divisor, bonus.
crit_chance = round(limb_damage * 20 + damage / divisor) + bonus * 10

# Dodge chance in percent of a creature (Azure Peak SIMPLEMOB_DODGE_*). weapon_skill is the attacker's rank with the weapon, fatigue the creature's dodge fatigue. Inputs: defender_speed, attacker_speed, weapon_skill, fatigue.
beast_dodge_chance = clamp(20 + (defender_speed - attacker_speed) * 3 - weapon_skill * 4 - fatigue, 5, 45)
